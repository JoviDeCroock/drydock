# Drydock — https://drydock.org
# Public marketing and documentation pages are indexable. The authenticated
# product (dashboard) and the JSON API are not useful in search results.
User-agent: *
Allow: /$
Allow: /docs
Allow: /privacy
Disallow: /dashboard
Disallow: /login
Disallow: /register
Disallow: /verify-email
Disallow: /api/
# A shared report link is a capability, not a public page. Indexing one would
# republish it to everyone long after the publisher revoked the share.
Disallow: /reports/
Disallow: /public/reports/
